/// A shape.
#[derive(Debug, Clone)]
pub enum Shape<'a> { Circle { r: f64 }, Named(&'a str) }

impl<'a> Shape<'a> {
    pub const fn new() -> Self { Shape::Circle { r: 1.0 } }
    fn area(&self) -> f64 {
        match self {
            Shape::Circle { r } => std::f64::consts::PI * r * r,
            Shape::Named(_) => 0.0,
        }
    }
}
