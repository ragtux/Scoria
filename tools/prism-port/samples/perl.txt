#!/usr/bin/perl
use strict;
use warnings;

# Count words
my %count;
my @lines = <STDIN>;
foreach my $line (@lines) {
    chomp $line;
    next if $line =~ /^\s*#/;
    $count{$_}++ for split /\s+/, $line;
}
my $total = scalar keys %count;
printf "%d words, first: %s\n", $total, $lines[0] // 'none';
