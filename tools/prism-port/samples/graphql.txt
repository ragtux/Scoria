# A query
query GetUser($id: ID!, $first: Int = 10) @cached(ttl: 60) {
  user(id: $id) {
    ...UserFields
    posts(first: $first, orderBy: {field: CREATED_AT, direction: DESC}) {
      edges { node { title body } }
    }
  }
}

fragment UserFields on User {
  id
  name @include(if: true)
}
