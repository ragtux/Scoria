# Server
user www-data;
worker_processes auto;

http {
    include       mime.types;
    server {
        listen 443 ssl http2;
        server_name example.com www.example.com;
        root /var/www/html;
        location ~* \.(png|jpg)$ {
            expires 30d;
        }
        location /api/ {
            proxy_pass http://127.0.0.1:8080;
            proxy_set_header Host $host;
            if ($request_method = POST) { return 405; }
        }
    }
}
