/// A counter.
@immutable
class Counter extends StatelessWidget with Mixin implements Comparable<Counter> {
  final int start;
  const Counter({Key? key, this.start = 0}) : super(key: key);

  @override
  Widget build(BuildContext context) => Text('count: $start ${start + 1}');

  int compareTo(Counter other) => start.compareTo(other.start);
}
