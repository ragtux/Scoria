require 'json'

module Shapes
  class Circle < Base
    include Comparable
    attr_reader :radius
    PI = 3.14159

    def initialize(radius = 1.0, **opts)
      @radius = radius
      @@count ||= 0
      $debug = opts.fetch(:debug, false)
    end

    def area = PI * @radius ** 2

    def to_s
      "Circle(#{@radius})"
    end

    def self.parse(str)
      return nil unless str =~ /\Acircle:(\d+)\z/i
      new($1.to_i)
    end
  end
end
