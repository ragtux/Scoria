(ns app.core
  (:require [clojure.string :as str]))

(defn greet
  "Greets someone."
  [name & {:keys [loud] :or {loud false}}]
  (let [msg (str "Hello, " name "!")]
    (if loud (str/upper-case msg) msg)))

(defmacro unless [test & body] `(if (not ~test) (do ~@body)))
