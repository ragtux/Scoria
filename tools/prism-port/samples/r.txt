# Load and summarize
library(dplyr)
df <- read.csv("data.csv", stringsAsFactors = FALSE)
f <- function(x, na.rm = TRUE) {
  if (is.null(x) || length(x) == 0L) return(NA)
  mean(x[!is.na(x)]) * 1e3 + 2i
}
result <- df %>% filter(age >= 18) %>% summarise(m = f(score))
